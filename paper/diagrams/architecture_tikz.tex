    \node[box=blue] (opt_stem) at (3.5, 4.5) {\textbf{Optical Stem}\\\scriptsize Conv3x3 $\to [B, 48, 128, 128]$};

    };

\caption{\textbf{CloudFree Vision v2 End-to-End Architecture.} The generator ingests four multi-modal sensor streams (LISS-IV optical, Sentinel-1 SAR, Sentinel-2 dry-season reference, and 4-channel CartoDEM topography). Spatial Transformer Networks (STN) ensure sub-pixel alignment before windowed multi-head cross-attention fusion. The decoder feeds dual output heads producing the clear-sky mean reconstruction $\hat{y}$ and heteroscedastic uncertainty variance $\log \sigma^2$.}
